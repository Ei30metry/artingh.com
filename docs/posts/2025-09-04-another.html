simplifyInfer :: Bool
              -> TcLevel
              -> [Sigma] -- TODO: Shouldn't this be a rho?
              -> WantedCts
              -> Tc ([(SkolTyVar, Kappa)], [Q])
simplifyInfer topLevelBD lvl phis wanteds = do
  traceTc "simplifyInfer" (indent 2 (pretty wanteds))
  solved <- runCs (solveWantedCts wanteds)
  wcsZonked <- zonkWantedCts solved
  traceTc "simplifyInfer - solved and zonked wanteds" (pretty wcsZonked)
  zphis <- mapM zonkSigma phis
  traceTc "simplifyInfer - zonked phis" (pretty zphis)
  if isInsolWC wcsZonked
    then do traceTc "simplifyInfer - insoluble" (pretty wcsZonked)
            throwError (InsolubleWC $ pretty wcsZonked)
    else do traceTc "simplifyInfer - not insoluble" (pretty wcsZonked)
            (qtvs, qs) <- decideQuantification topLevelBD zphis
            emitResidualCts lvl (map fst qtvs) qs wcsZonked
            pure (qtvs, qs)

simplifyTopWantedCts :: WantedCts -> Cs WantedCts
simplifyTopWantedCts = nestCs . solveWantedCts

simplifyTop :: WantedCts -> Tc ()
simplifyTop wcs = do
  traceTc "simplifyTop - start" (indent 2 (pretty wcs))
  final_wc  <- runCs (simplifyTopWantedCts wcs)
  if isEmptyWC final_wc
    then traceTc "simplifyTop - end" emptyDoc
    else throwError (InsolubleWC (pretty final_wc))

pushLevelAndSolveEqsX :: Tc a -> Tc (TcLevel, WantedCts, a)
pushLevelAndSolveEqsX tc = do
  lvl <- getTcLevel
  let plvl = pushTcLevel lvl
  (wanted, res) <- updEnv (setAmbientTcLevel plvl) $ do
    (res, wanted) <- captureCts tc
    wanted'       <- runCs (do traceCs "pushLevelAndSolveEqsX - start" $
                                 vsep [pretty "plvl   =" <+> pretty plvl
                                      ,pretty "wanted =" <+> pretty wanted]
                               solveWantedCts wanted)

    traceTc "pushLevelAndSolveEqsX - end" $
      vsep [pretty "wanted':" <+> pretty wanted'
           ,pretty "res:    " <+> emptyDoc]
    return (wanted', res)
  return (plvl, wanted, res)

solveEquality :: CtFlavour -> TcSigma -> TcSigma -> Cs ()
solveEquality flv lhs rhs = do
  r_lhs <- rewriteSigma lhs
  r_rhs <- rewriteSigma rhs
  traceCs "solveEquality - zonked types" $
    vsep [pretty "LHS:" <+> pretty r_lhs
         ,pretty "RHS:" <+> pretty r_rhs]
  (irreds, canons) <- canonEqSigma flv r_lhs r_rhs
  traceCs "solveEquality - constraints" $
    vsep [pretty "Canonicals:"   <+> align (pprListV canons)
         ,pretty "Irreducibles:" <+> align (pprListV irreds)]
  mapM_ updInertIrreds irreds
  -- See Note [The relation between unification and givens]
  updInertEqsOrUnify canons
